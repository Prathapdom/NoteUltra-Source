cmake_minimum_required(VERSION 3.20)
project(NoteUltra VERSION 1.0.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(NoteUltra WIN32 main.cpp app.rc)
target_include_directories(NoteUltra PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(NoteUltra PRIVATE comctl32 comdlg32 ole32 uuid shell32 gdi32 uxtheme gdiplus dwmapi)
if(MINGW)
 target_link_options(NoteUltra PRIVATE -municode -static)
else()
 target_compile_options(NoteUltra PRIVATE /utf-8 /W4)
 set_property(TARGET NoteUltra PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
foreach(dependency Scintilla Lexilla)
 add_custom_command(TARGET NoteUltra POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/vendor/wscite/wscite/${dependency}.dll" "$<TARGET_FILE_DIR:NoteUltra>/${dependency}.dll")
endforeach()
add_custom_command(TARGET NoteUltra POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/vendor/scintilla/License.txt" "$<TARGET_FILE_DIR:NoteUltra>/Scintilla-LICENSE.txt" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/vendor/lexilla/License.txt" "$<TARGET_FILE_DIR:NoteUltra>/Lexilla-LICENSE.txt")
enable_testing()
add_executable(io-tests io-tests.cpp)
add_test(NAME file-io COMMAND io-tests)
add_executable(smoke-tests smoke-tests.cpp)
target_link_libraries(smoke-tests PRIVATE comctl32)
if(MINGW)
 target_link_options(io-tests PRIVATE -static)
 target_link_options(smoke-tests PRIVATE -municode -static)
else()
 target_compile_options(smoke-tests PRIVATE /utf-8)
endif()
add_test(NAME release-workflow COMMAND smoke-tests "$<TARGET_FILE:NoteUltra>")
